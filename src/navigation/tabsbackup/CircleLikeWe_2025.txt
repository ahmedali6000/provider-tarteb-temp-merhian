  export const HomeTabs = () => {

        let icon = '';

        switch (routeName) {
          case 'HomeScreen':
            icon = 'home';
            newColor= '#ddd'
            break;
            case 'FavouriteScreen':

            newColor= 'red'
            break;

            icon = 'person';
            newColor= '#9cfff8'
            break;
            case 'OrderStack':
            icon = 'briefcase';
            newColor= 'yellow'
            break;
            
            
        }
    
        return (
          <Ionicons
            name={icon}
            size={25}
            color={routeName === selectedTab ? newColor : 'white'}
          />
        );
      };
      const renderTabBar = ({ routeName, selectedTab, navigate }: any) => {
        return (
          <TouchableOpacity
            onPress={() => navigate(routeName)}
            style={{
              flex: 1,
              alignItems: 'center',
              justifyContent: 'center',
            }}>
            {_renderIcon(routeName, selectedTab)}
          </TouchableOpacity>
        );
      };
    
      return (
        <View style={{ flex: 1 }}>
          <CurvedBottomBar.Navigator

            <CurvedBottomBar.Screen
              options={{ headerShown: false }}

              position="LEFT"
              component={HomeScreen}
            />

              options={{ headerShown: false , unmountOnBlur:true }}
